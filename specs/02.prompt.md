# Refinitiv (LSEG) Stock Price Provider Implementation

**Feature Branch:** feature/lseg-stock-prices

## Objective
Implement stock price data retrieval from Refinitiv (LSEG) as an alternative provider to IBKR, accessible via explicit provider selection parameter. Supports real-time quotes and historical price data with intelligent RIC conversion, session management, caching, and retry logic.

## Core Requirements

### 1. Provider Selection Strategy
**Decision:** Explicit Selection via Query Parameter

- Keep IBKR as default provider (no breaking changes)
- Add `?provider=lseg` query parameter to API endpoints
- User explicitly chooses Refinitiv when needed
- Zero impact on existing IBKR functionality

**API Examples:**

### 4. Ticker-to-RIC Conversion
**Decision:** Cache with 24h TTL

**Strategy:**
1. Check in-memory cache first (O(1) lookup)
2. If cache miss or expired, search via Refinitiv API
3. Cache result with 24-hour TTL
4. Cache stored in dictionary: `{ticker: (ric, timestamp)}`

**Implementation:**
```python
class LSEGProvider:
    def __init__(self):
        self._ric_cache: Dict[str, Tuple[str, datetime]] = {}
        self._cache_ttl = timedelta(hours=24)

    def _ticker_to_ric(self, ticker: str) -> str:
        """Convert ticker to RIC with caching."""
        # Check cache
        if ticker in self._ric_cache:
            ric, cached_at = self._ric_cache[ticker]
            if datetime.now(timezone.utc) - cached_at < self._cache_ttl:
                return ric

        # Cache miss - search API
        response = search.Definition(
            query=ticker,
            view=search.Views.EQUITY_INSTRUMENTS,
            select="RIC, TickerSymbol",
            top=1
        ).get_data()

        if response.data.df is None or response.data.df.empty:
            raise DataNotFoundError("Ticker", ticker,
                details={"reason": "No RIC found for ticker"})

        ric = response.data.df['RIC'].iloc[0]
        self._ric_cache[ticker] = (ric, datetime.now(timezone.utc))
        return ric
```

### 5. Error Handling & Retry Logic
**Decision:** Retry 3x with Exponential Backoff

**Retry Strategy:**
- Retry on: Connection errors, 429 rate limit, 5xx server errors
- Don't retry on: 404 not found, 401 auth failure, 400 bad request
- Backoff: 1 second, 2 seconds, 4 seconds (exponential)
- After 3 failures: Raise exception with details

**Implementation:**
```python
import time
from typing import TypeVar, Callable

T = TypeVar('T')

def _retry_with_backoff(self, func: Callable[..., T], *args, **kwargs) -> T:
    """Execute function with retry logic and exponential backoff."""
    max_retries = 3
    base_delay = 1.0

    for attempt in range(max_retries):
        try:
            return func(*args, **kwargs)
        except Exception as e:
            # Check if retryable error
            if not self._is_retryable_error(e):
                raise

            if attempt < max_retries - 1:
                delay = base_delay * (2 ** attempt)
                logger.warning(f"Attempt {attempt + 1} failed: {e}. Retrying in {delay}s...")
                time.sleep(delay)
            else:
                logger.error(f"All {max_retries} attempts failed")
                raise ProviderConnectionError(
                    f"Refinitiv API failed after {max_retries} retries",
                    details={"error": str(e), "attempts": max_retries}
                )

def _is_retryable_error(self, error: Exception) -> bool:
    """Check if error should trigger retry."""
    error_str = str(error).lower()
    # Retry on connection, timeout, rate limit, server errors
    return any(keyword in error_str for keyword in [
        "timeout", "connection", "429", "500", "502", "503"
    ])
```

### 6. Session Management
**Decision:** Smart Refresh (Lazy + Proactive)

**Session Lifecycle:**
- Create session on first request (lazy initialization)
- Session valid for 1 hour (Refinitiv default)
- Refresh proactively at 55 minutes (before expiry)
- Auto-recreate on 401 authentication errors
- Track session creation timestamp

**Implementation:**
```python
class LSEGProvider:
    def __init__(self):
        self._session = None
        self._session_created_at: Optional[datetime] = None
        self._session_ttl = timedelta(minutes=55)  # Refresh before 1h expiry

    def _get_session(self):
        """Get or create Refinitiv session with smart refresh."""
        now = datetime.now(timezone.utc)

        # Check if session needs refresh
        if self._session is None or self._session_created_at is None:
            needs_refresh = True
        elif now - self._session_created_at > self._session_ttl:
            needs_refresh = True
            logger.info("Session TTL exceeded, refreshing...")
        else:
            needs_refresh = False

        if needs_refresh:
            self._create_session()

        return rd.session.default

    def _create_session(self):
        """Create new Refinitiv session."""
        try:
            session = rd.session.platform.Definition(
                app_key=settings.LSEG_APP_KEY,
                grant=rd.session.platform.GrantPassword(
                    username=settings.LSEG_USERNAME,
                    password=settings.LSEG_PASSWORD
                ),
                signon_control=True
            ).get_session()

            session.open()
            rd.session.set_default(session)

            self._session = session
            self._session_created_at = datetime.now(timezone.utc)
            logger.info("Refinitiv session created successfully")

        except Exception as e:
            logger.error(f"Failed to create Refinitiv session: {e}")
            raise ProviderConnectionError(
                "Could not authenticate with Refinitiv",
                details={"error": str(e)}
            )
```

```python
def _validate_and_extract(self, df: pd.DataFrame, symbol: str) -> PriceQuote:
    """Extract and validate quote data from DataFrame."""
    if df is None or df.empty:
        raise DataNotFoundError("Quote", symbol)

    row = df.iloc[0]

    # Extract fields with safe handling
    def safe_decimal(field_name: str, required: bool = False) -> Optional[Decimal]:
        if field_name not in df.columns:
            if required and settings.LSEG_STRICT_VALIDATION:
                raise DataValidationError(
                    f"Required field '{field_name}' missing for {symbol}"
                )
            return None

        value = row[field_name]
        if pd.isna(value):
            if required and settings.LSEG_STRICT_VALIDATION:
                raise DataValidationError(
                    f"Required field '{field_name}' is NaN for {symbol}"
                )
            return None

        return Decimal(str(value))

    # Build PriceQuote
    return PriceQuote(
        symbol=symbol,
        last_price=safe_decimal("TRDPRC_1", required=True),  # Always required
        bid=safe_decimal("BID"),
        ask=safe_decimal("ASK"),
        open=safe_decimal("OPEN_PRC"),
        high=safe_decimal("HIGH_1"),
        low=safe_decimal("LOW_1"),
        close=safe_decimal("CLOSE"),
        volume=int(row["ACVOL_1"]) if "ACVOL_1" in df.columns and pd.notna(row["ACVOL_1"]) else None,
        timestamp=datetime.now(timezone.utc)
    )
```

### 8. Performance Optimization
**Decision:** Combined Throttling + Caching

**Configuration:**
```python
# app/config.py
class Settings(BaseSettings):
    # ... existing settings ...

    # LSEG Performance
    LSEG_RATE_LIMIT_PER_SEC: int = Field(
        default=10,
        description="Maximum API requests per second to Refinitiv"
    )

    LSEG_CACHE_TTL_SECONDS: int = Field(
        default=5,
        description="Cache quote data for X seconds (0 = disabled)"
    )
```

**Throttling Implementation:**
```python
import asyncio
from collections import deque

class LSEGProvider:
    def __init__(self):
        self._request_times: deque = deque(maxlen=settings.LSEG_RATE_LIMIT_PER_SEC)

    def _throttle(self):
        """Ensure rate limit is respected."""
        now = time.time()

        # Remove timestamps older than 1 second
        while self._request_times and now - self._request_times[0] > 1.0:
            self._request_times.popleft()

        # If at rate limit, wait
        if len(self._request_times) >= settings.LSEG_RATE_LIMIT_PER_SEC:
            sleep_time = 1.0 - (now - self._request_times[0])
            if sleep_time > 0:
                time.sleep(sleep_time)

        self._request_times.append(time.time())
```

**Caching Implementation:**
```python
from typing import Optional, Tuple

class LSEGProvider:
    def __init__(self):
        self._quote_cache: Dict[str, Tuple[PriceQuote, datetime]] = {}

    def _get_cached_quote(self, symbol: str) -> Optional[PriceQuote]:
        """Get cached quote if still valid."""
        if settings.LSEG_CACHE_TTL_SECONDS <= 0:
            return None

        if symbol in self._quote_cache:
            quote, cached_at = self._quote_cache[symbol]
            age = (datetime.now(timezone.utc) - cached_at).total_seconds()

            if age < settings.LSEG_CACHE_TTL_SECONDS:
                logger.debug(f"Cache hit for {symbol} (age: {age:.1f}s)")
                return quote

        return None

    def _cache_quote(self, symbol: str, quote: PriceQuote):
        """Cache quote data."""
        if settings.LSEG_CACHE_TTL_SECONDS > 0:
            self._quote_cache[symbol] = (quote, datetime.now(timezone.utc))
```
